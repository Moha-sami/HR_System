Table Component

Reusable data table with variants, sizes, sorting, alignment, and custom cell templates.

Selector

<app-table [columns]="columns" [rows]="rows" />

Basic Usage

<!-- Component -->
columns: ColumnDef[] = [
  {{ '{' }} key: 'label', label: 'Detail', align: 'left' {{ '}' }},
  {{ '{' }} key: 'value', label: 'Amount', align: 'right' {{ '}' }},
];

rows = [
  {{ '{' }} label: 'Basic Salary', value: '5,000.00' {{ '}' }},
  {{ '{' }} label: 'Housing Allowance', value: '1,500.00' {{ '}' }},
];

<!-- Template -->
<app-table [columns]="columns" [rows]="rows" />

Column Alignment

columns: ColumnDef[] = [
  {{ '{' }} key: 'label', label: 'Detail', align: 'left' {{ '}' }},
  {{ '{' }} key: 'value', label: 'Amount', align: 'right' {{ '}' }},
];

// Supported: 'left' | 'center' | 'right'

Custom Column Widths

columns: ColumnDef[] = [
  {{ '{' }} key: 'code', label: 'Code', width: '100px' {{ '}' }},
  {{ '{' }} key: 'name', label: 'Product', width: '40%' {{ '}' }},
  {{ '{' }} key: 'price', label: 'Price', width: '100px' {{ '}' }},
];

// Width accepts CSS values: '100px', '40%', '1fr', etc.
// Columns without width share remaining space equally (1fr)

Sorting

@if (lastSortChange) {
Sorted by {{ lastSortChange.column }} {{ lastSortChange.direction === 'asc' ? '↑ ascending' : '↓ descending' }}
} 
columns: ColumnDef[] = [
  {{ '{' }} key: 'name', label: 'Name', sortable: true {{ '}' }},
  {{ '{' }} key: 'department', label: 'Dept', sortable: false {{ '}' }},
];

// sortable: true  → shows sort icon, click to sort
// sortable: false → no sort icon, click does nothing
// undefined       → same as false

<app-table
  [columns]="columns"
  [rows]="rows"
  (sortChange)="onSort($event)"
/>

// Event: {{ '{' }} column: string; direction: 'asc' | 'desc' {{ '}' }}

Empty State

<app-table
  [columns]="columns"
  [rows]="[]"
  emptyMessage="No data available"
/>

Row Click Event

@if (lastClickedRow) {
Clicked: {{ lastClickedRow.name }} — {{ lastClickedRow.position }}
} 
<app-table
  [columns]="columns"
  [rows]="rows"
  (rowClick)="onRowClick($event)"
/>

onRowClick(row: any) {{ '{' }}
  console.log('Clicked:', row);
{{ '}' }}

Custom Cell Templates

Render images, badges, buttons, or any custom content in cells using ng-template.

{{ row.name }}
{{ row.status }} 
<!-- 1. Define templates -->
<ng-template #avatarTemplate let-row>
  <div class="flex items-center gap-2">
    <img [src]="row.avatar" class="w-8 h-8 rounded-full" />
    <span>{{ '{{' }} row.name {{ '}}' }}</span>
  </div>
</ng-template>

<ng-template #statusTemplate let-row>
  <span class="px-2 py-1 text-xs rounded-full bg-green-100 text-green-700">
    {{ '{{' }} row.status {{ '}}' }}
  </span>
</ng-template>

<ng-template #actionsTemplate let-row>
  <button (click)="edit(row)">Edit</button>
  <button (click)="delete(row)">Delete</button>
</ng-template>

<!-- 2. Collect templates in component -->
@ViewChild('avatarTemplate') avatarTemplate!: TemplateRef<CellContext>;
customTemplates = signal<Map<string, TemplateRef<CellContext>>>(new Map());

ngAfterViewInit(): void {{ '{' }}
  const map = new Map();
  map.set('avatarTemplate', this.avatarTemplate);
  map.set('statusTemplate', this.statusTemplate);
  map.set('actionsTemplate', this.actionsTemplate);
  this.customTemplates.set(map);
{{ '}' }}

<!-- 3. Reference in column def -->
columns: ColumnDef[] = [
  {{ '{' }} key: 'name', label: 'User', template: 'avatarTemplate' {{ '}' }},
  {{ '{' }} key: 'status', label: 'Status', template: 'statusTemplate' {{ '}' }},
  {{ '{' }} key: 'actions', label: 'Actions', template: 'actionsTemplate' {{ '}' }},
];

<!-- 4. Pass to table -->
<app-table
  [columns]="columns"
  [rows]="rows"
  [cellTemplates]="customTemplates()"
/>

API Reference

Inputs

Input Type Default Description
columns ColumnDef[] [] Column definitions
rows any[] [] Row data
variant 'default' | 'striped' | 'bordered' 'default' Visual style
size 'sm' | 'md' | 'lg' 'md' Row height and padding
emptyMessage string 'No data available' Message when rows is empty
cellTemplates Map<string, TemplateRef> new Map() Custom cell templates

Outputs

Output Type Description
rowClick any Emits clicked row object
sortChange {{ '{' }} column: string; direction: 'asc' | 'desc' {{ '}' }} Emits sort state changes

ColumnDef Interface

interface ColumnDef {{ '{' }}
  key: string;              // Property name in row object
  label: string;            // Header text
  align?: 'left' | 'center' | 'right';
  width?: string;           // CSS width ('100px', '40%', etc.)
  sortable?: boolean;       // Enable sorting on this column
  template?: string;        // Template reference name for custom cells
{{ '}' }}
